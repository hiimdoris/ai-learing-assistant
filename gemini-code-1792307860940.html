// Google Apps Script 呼叫 Gemini 範例 function GEMINI_QUIZ(textPrompt) { const apiKey = PropertiesService.getScriptProperties().getProperty('GOOGLE_API_KEY'); const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`; const payload = { "contents": [{ "parts": [{ "text": "請根據以下內容出 3 道選擇題，專有名詞請保留英文：\n\n" + textPrompt }] }] }; const options = { "method": "post", "contentType": "application/json", "payload": JSON.stringify(payload) }; const response = UrlFetchApp.fetch(url, options); const json = JSON.parse(response.getContentText()); return json.candidates[0].content.parts[0].text; }
